from collections import deque

def water_jug(a, b, target):
    q = deque([(0, 0)])
    visited = set()

    while q:
        x, y = q.popleft()

        if (x, y) in visited:
            continue

        visited.add((x, y))
        print(x, y)

        if x == target or y == target:
            print("Target reached!")
            return

        q.extend([
            (a, y),       # Fill jug A
            (x, b),       # Fill jug B
            (0, y),       # Empty jug A
            (x, 0),       # Empty jug B
            (max(0, x-(b-y)), min(b, x+y)),
            (min(a, x+y), max(0, y-(a-x)))
        ])

    print("Solution not possible")

water_jug(4, 3, 2)
