def minimax(depth, is_max):
    if depth == 3:
        return values.pop(0)

    if is_max:
        return max(minimax(depth + 1, False),
                   minimax(depth + 1, False))
    else:
        return min(minimax(depth + 1, True),
                   minimax(depth + 1, True))

values = [3, 5, 2, 9, 12, 5, 23, 23]

print("Best value:", minimax(0, True))
