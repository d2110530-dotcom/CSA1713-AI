graph = {
    'A': [('B', 1), ('C', 4)],
    'B': [('D', 2)],
    'C': [('D', 1)],
    'D': []
}

h = {'A': 3, 'B': 2, 'C': 1, 'D': 0}

start = 'A'
goal = 'D'
open_list = [(h[start], start)]
cost = {start: 0}
parent = {}

while open_list:
    open_list.sort()
    f, node = open_list.pop(0)

    if node == goal:
        break

    for neighbor, weight in graph[node]:
        new_cost = cost[node] + weight

        if neighbor not in cost or new_cost < cost[neighbor]:
            cost[neighbor] = new_cost
            parent[neighbor] = node
            open_list.append((new_cost + h[neighbor], neighbor))

path = []
node = goal

while node != start:
    path.append(node)
    node = parent[node]

path.append(start)
path.reverse()

print("Path:", path)
print("Cost:", cost[goal])
